**네, 구글 시트의 특정 탭(시트) 이름을 지정해서 명단을 가져오거나 출결 데이터를 기록하도록 수정할 수 있습니다!** 이렇게 수정하면 **1반, 2반, 3반 등 탭별로 명단을 따로 관리**하거나, 특정 탭(예: `학생명단`)에서만 이름을 불러오는 것이 가능해집니다. --- ### 1. Google Apps Script 코드 수정 (`Code.gs`) Apps Script에서 탭 이름(`tab`) 매개변수를 받아 해당 탭을 열도록 변경합니다. 탭 이름을 전달하지 않거나 찾을 수 없으면 **첫 번째 탭**을 기본으로 가져옵니다. 스프레드시트의 **[확장 프로그램] > [Apps Script]**에서 기존 코드를 아래 코드로 교체하신 후 **`[배포] > [새 배포]`**를 진행해 주세요. ```javascript // 탭 이름을 찾아오거나 없으면 첫 번째 탭 반환하는 함수 function getTargetSheet(ss, sheetName) { if (sheetName) { var sheet = ss.getSheetByName(sheetName); if (sheet) return sheet; } return ss.getSheets(); // 기본값: 첫 번째 탭 } // 1. 구글 시트의 특정 탭에서 학생 명단 읽어오기 (doGet) function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // URL 매개변수로 ?tab=탭이름 형태로 들어오면 해당 탭을 읽음 var targetTab = (e && e.parameter && e.parameter.tab) ? e.parameter.tab : ""; var sheet = getTargetSheet(ss, targetTab); var data = sheet.getDataRange().getValues(); var studentList = []; for (var i = 0; i < data.length; i++) { var num = parseInt(data[i], 10); // A열: 번호 var name = String(data[i] || "").trim(); // B열: 이름 if (!isNaN(num) && name) { studentList.push({ id: num, name: name }); } } return ContentService.createTextOutput(JSON.stringify(studentList)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ error: error.message })) .setMimeType(ContentService.MimeType.JSON); } } // 2. 구글 시트의 특정 탭에 출석 데이터 기록하기 (doPost) function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); // 요청에 포함된 tabName 탭에 기록 (없으면 첫 번째 탭) var targetTab = data.tabName || ""; var sheet = getTargetSheet(ss, targetTab); if (sheet.getLastRow() === 0) { sheet.appendRow(["전송 일시", "학급 이름", "번호", "이름", "출석 상태", "출석 시간"]); } var today = Utilities.formatDate(new Date(), "GMT+9", "yyyy-MM-dd"); var className = data.className || "미지정 학급"; if (data.records && Array.isArray(data.records)) { data.records.forEach(function(item) { sheet.appendRow([today, className, item.id + "번", item.name, item.status, item.time]); }); } return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT); } catch(error) { return ContentService.createTextOutput("Error: " + error.message).setMimeType(ContentService.MimeType.TEXT); } } ``` --- ### 2. 앱 코드 수정 (`index.html`) 컨트롤 패널에 **`🏷️ 탭 이름`** 입력칸을 추가하여 원하는 탭 이름을 지정할 수 있도록 개선한 전체 코드입니다. ```html 학생 출석 체크 시스템
